## 解决方案：使用 Cubism SDK 直接实现

### 问题分析

live2d-widget 库存在元素未找到的问题，导致模型无法显示。使用 Cubism SDK 直接实现可以避免这个问题，提供更直接、更可控的模型加载和渲染过程。

### 技术方案

**步骤 1：修改 Live2DCharacter.vue 组件**

**操作**：
- 编辑 `Live2DCharacter.vue` 文件
- 移除 live2d-widget 相关代码
- 添加 Cubism SDK 直接实现

**修改内容**：

1. **替换初始化方法**：
   - 移除 `initLive2D` 方法中的 live2d-widget 相关代码
   - 添加 Cubism SDK 直接初始化代码

2. **实现模型加载逻辑**：
   - 加载 Cubism SDK 脚本
   - 解析模型文件
   - 加载必要的资源文件（moc3、纹理等）
   - 初始化渲染器
   - 开始动画循环

3. **添加错误处理**：
   - 捕获并处理模型加载过程中的错误
   - 提供清晰的错误信息

4. **实现清理方法**：
   - 释放模型和渲染器资源
   - 停止动画循环

async initLive2D() {
  try {
    this.isLoading = true;

    console.log('=== 开始初始化Live2D ===');
    
    // 检查模型文件是否存在
    console.log('检查模型文件:', this.modelPath);
    const modelResponse = await fetch(this.modelPath);
    
    if (!modelResponse.ok) {
      throw new Error(`模型文件加载失败: ${modelResponse.status}`);
    }
    
    const modelSettings = await modelResponse.json();
    console.log('模型文件加载成功');
    
    // 加载 Cubism SDK
    if (!window.live2d) {
      console.log('加载 Cubism SDK...');
      await this.loadScript('https://cdn.jsdelivr.net/npm/cubism-sdk-for-web@4.2.0/dist/cubism.core.min.js');
      await this.loadScript('https://cdn.jsdelivr.net/npm/cubism-sdk-for-web@4.2.0/dist/cubism.framework.min.js');
      console.log('Cubism SDK 加载完成');
    }
    
    // 清空容器
    const container = this.$refs.live2dContainer;
    container.innerHTML = '';
    
    // 创建 canvas 元素
    const canvas = document.createElement('canvas');
    canvas.width = 200;
    canvas.height = 300;
    container.appendChild(canvas);
    console.log('创建 canvas 元素:', canvas);
    
    // 初始化 Cubism SDK
    console.log('初始化 Cubism SDK...');
    
    // 加载 moc3 文件
    const mocPath = `/live2d-models/hiyori_pro_zh/${modelSettings.FileReferences.Moc}`;
    console.log('加载 moc3 文件:', mocPath);
    const mocResponse = await fetch(mocPath);
    
    if (!mocResponse.ok) {
      throw new Error(`moc3 文件加载失败: ${mocResponse.status}`);
    }
    
    const mocArrayBuffer = await mocResponse.arrayBuffer();
    console.log('moc3 文件加载成功');
    
    // 加载纹理
    console.log('加载纹理文件...');
    const textures = [];
    
    for (let i = 0; i < modelSettings.FileReferences.Textures.length; i++) {
      const texturePath = `/live2d-models/hiyori_pro_zh/${modelSettings.FileReferences.Textures[i]}`;
      console.log(`加载纹理 ${i}:`, texturePath);
      
      const texture = await this.loadTexture(texturePath);
      textures.push(texture);
      console.log(`纹理 ${i} 加载成功`);

    // 初始化渲染器
    console.log('初始化渲染器...');
    const gl = canvas.getContext('webgl') || canvas.getContext('experimental-webgl');
    
    if (!gl) {
      throw new Error('浏览器不支持 WebGL');
    }
    
    // 这里需要实现 Cubism SDK 的具体初始化逻辑

    this.isLoading = false;
    
  } catch (err) {
    console.error('=== 初始化Live2D失败 ===', err);
    this.error = `初始化失败: ${err.message}`;
    this.isLoading = false;
  }
},

loadTexture(url) {
  return new Promise((resolve, reject) => {
    const image = new Image();
    image.crossOrigin = 'anonymous';
    image.onload = () => {
      console.log('纹理加载完成:', url);
      resolve(image);
    };
    image.onerror = () => {
      console.error('纹理加载失败:', url);
      reject(new Error(`纹理加载失败: ${url}`));
    };
    image.src = url;
  });
},
